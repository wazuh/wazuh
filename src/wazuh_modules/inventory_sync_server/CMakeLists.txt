cmake_minimum_required(VERSION 3.12.4)

project(inventory_sync_server CXX)

# Colocate the .so next to the other manager shared libs (e.g. libvulnerability_scanner.so)
# so modulesd's runtime rpath ($ORIGIN/../lib) resolves it.
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)

file(GLOB_RECURSE INVENTORY_SYNC_SERVER_SRC "src/*.cpp")

add_library(inventory_sync_server SHARED ${INVENTORY_SYNC_SERVER_SRC})

# FlatBuffers schema: the FullSession contract IS the single shared inventorySync.fbs
# (shared_modules/utils/flatbuffers/schemas), generated by the compile_schemas_input target VD's
# database_feed already depends on -- see src/schema/syncSchema.hpp.
if(TARGET compile_schemas_input)
  add_dependencies(inventory_sync_server compile_schemas_input)
endif()

target_include_directories(
  inventory_sync_server
  PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include
         # Shared generated schema header + the header-only flatbuffers runtime it includes.
         # PUBLIC so the unit-test target inherits both through target_link_libraries.
         ${SRC_FOLDER}/shared_modules/utils
         ${SRC_FOLDER}/external/flatbuffers/include
  # src/ is on the path so submodules resolve by prefix: "http_server/...", "endpoints/...",
  # "common/...".
  PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src
          ${SRC_FOLDER}/shared_modules/utils # loggerHelper.h, singleton.hpp, proc.hpp
          ${SRC_FOLDER}/shared_modules/common # commonDefs.h
          ${SRC_FOLDER}/external/cJSON # cJSON.h (pulled by commonDefs.h)
          ${SRC_FOLDER}/external/nlohmann # json.hpp
          # ONLY for src/vd/vdScannerAdapter.cpp: the scanner facade's header pulls headers
          # spelled relative to the source root ("external/nlohmann/json.hpp"). Contained to
          # that one TU (see vd/vdScannerFactory.hpp).
          ${SRC_FOLDER})

# The HTTP-over-UDS transport is the shared wazuh_uds_http_server archive (extracted from this
# module): asio/llhttp stay PRIVATE inside it and never leak here. Guarded add_subdirectory so
# this module still configures standalone.
if(NOT TARGET wazuh_uds_http_server)
  add_subdirectory(${SRC_FOLDER}/shared_modules/uds_http_server
                   ${CMAKE_BINARY_DIR}/shared_modules/uds_http_server EXCLUDE_FROM_ALL)
endif()

# wazuhext provides cJSON (bundled whole-archive, see external/CMakeLists.txt): the extern "C" shim
# calls cJSON_Print/cJSON_free to convert the borrowed <indexer> subtree. Taking it from the shared
# library rather than linking ext_cjson statically avoids a second copy of cJSON inside this .so,
# which would sit alongside the one modulesd already has loaded.
# vulnerability_scanner: the VD scan lane calls its neutral entry points (runScannerFromViews,
# the vd_sync coordination registry). The dependency is one-way -- the scanner never includes a
# header of this module -- so no cycle: the scanner's own coordination needs are served through
# vd_sync::IScanCoordinator implementations REGISTERED by the sync modules.
# wazuh_metrics is PUBLIC: the D18 instrumentation lives in headers under src/ that the unit-test
# target compiles too, so its include dirs must propagate through target_link_libraries.
target_link_libraries(
  inventory_sync_server
  PUBLIC wazuh_metrics
  PRIVATE pthread wazuh_uds_http_server wazuhext indexer_connector vulnerability_scanner)

set_target_properties(inventory_sync_server PROPERTIES BUILD_RPATH_USE_ORIGIN TRUE
                                                      BUILD_RPATH "$ORIGIN/../lib")

if(UNIT_TEST)
  enable_testing()
  add_coverage_libraries(inventory_sync_server)
  add_subdirectory(test)
endif()

# The VD integration workflow's driver. Unconditional, like the legacy module's testtool was: the
# workflow builds exactly this target without turning unit tests on.
add_subdirectory(testtool)
